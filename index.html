// ========================================== // BACKEND: MENERIMA SCAN BARCODE & FOTO (LOG_HISTORY A-D) // ========================================== function doPost(e) { try { var data = JSON.parse(e.postData.contents); var barcode = data.barcode; var base64Image = data.image; var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheetName = "LOG_HISTORY"; var sheet = ss.getSheetByName(sheetName); if (!sheet) { sheet = ss.insertSheet(sheetName); sheet.appendRow(["Waktu", "Data Barcode", "Keterangan", "Foto Bukti"]); sheet.getRange("A1:D1").setFontWeight("bold"); } var timestamp = Utilities.formatDate(new Date(), "GMT+7", "dd/MM/yyyy HH:mm:ss"); var keterangan = "Scan Otomatis via VSC BS-8669"; var linkFoto = "Tidak ada foto"; if (base64Image && base64Image.indexOf("data:image") > -1) { var folderId = "1JvaaXyTYEQ0hFPCQ5gxRga7Wnmx4N8bK"; var folder = DriveApp.getFolderById(folderId); var splitData = base64Image.split(","); var contentType = splitData[0].match(/:(.*?);/)[1]; var rawBytes = Utilities.base64Decode(splitData[1]); var namaFile = "Scan_" + barcode + "_" + timestamp.replace(/[\/:\s]/g, "-") + ".jpg"; var blob = Utilities.newBlob(rawBytes, contentType, namaFile); var fileBaru = folder.createFile(blob); linkFoto = fileBaru.getUrl(); } sheet.appendRow([timestamp, barcode, keterangan, linkFoto]); return ContentService.createTextOutput(JSON.stringify({status: "success"})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } // ========================================== // BACKEND: SISTEM ROUTING UNTUK AMBIL DATA (GET) // ========================================== function doGet(e) { var action = e && e.parameter && e.parameter.action; // Jalur Web Laporan (LOG_HISTORY Kolom F-J) if (action === "laporan") { return ambilDataLaporan(); } // Jalur Web Presensi (KEHADIRAN Kolom A-C) if (action === "presensi") { return ambilDataPresensi(); } return ContentService.createTextOutput("API Aktif. Gunakan parameter ?action=laporan atau ?action=presensi"); } function ambilDataLaporan() { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("LOG_HISTORY"); if (!sheet) return buatResponJSON({ error: "Sheet LOG_HISTORY tidak ditemukan" }); var values = sheet.getDataRange().getDisplayValues(); var dataPilihan = values.map(function(row) { return [row[5] || "", row[6] || "", row[7] || "", row[8] || "", row[9] || ""]; }).filter(function(row) { return row.some(function(cell) { return cell !== ""; }); }); return buatResponJSON(dataPilihan); } catch (err) { return buatResponJSON({ error: err.toString() }); } } function ambilDataPresensi() { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("KEHADIRAN"); if (!sheet) return buatResponJSON({ error: "Sheet KEHADIRAN tidak ditemukan" }); var values = sheet.getDataRange().getDisplayValues(); var dataPilihan = values.map(function(row) { return [row[0] || "", row[1] || "", row[2] || ""]; }).filter(function(row) { return row.some(function(cell) { return cell !== ""; }); }); return buatResponJSON(dataPilihan); } catch (err) { return buatResponJSON({ error: err.toString() }); } } function buatResponJSON(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
